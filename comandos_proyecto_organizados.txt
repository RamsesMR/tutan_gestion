8. CONSULTAS DE COMPROBACIÓN EN POSTGRESQL
============================================================

IMPORTANTE:
Los siguientes bloques son SQL. Se ejecutan en PostgreSQL, no en PowerShell.

# Comprobar cobertura horaria

SELECT
    mercados.simbolo_proveedor AS mercado,
    COUNT(*) AS total_velas,
    COUNT(DISTINCT velas.fecha_apertura) AS fechas_distintas,
    MIN(velas.fecha_apertura) AS primera_vela,
    MAX(velas.fecha_apertura) AS ultima_vela
FROM velas
INNER JOIN mercados
    ON mercados.id = velas.mercado_id
WHERE velas.intervalo = '1h'
GROUP BY mercados.simbolo_proveedor
ORDER BY mercados.simbolo_proveedor;


# Comprobar cobertura histórica completa

SELECT
    mercados.simbolo_proveedor AS mercado,
    velas.intervalo,
    COUNT(*) AS total_velas,
    COUNT(DISTINCT velas.fecha_apertura) AS fechas_distintas,
    MIN(velas.fecha_apertura) AS primera_vela,
    MAX(velas.fecha_apertura) AS ultima_vela
FROM velas
INNER JOIN mercados
    ON mercados.id = velas.mercado_id
GROUP BY
    mercados.simbolo_proveedor,
    velas.intervalo
ORDER BY
    mercados.simbolo_proveedor,
    velas.intervalo;
